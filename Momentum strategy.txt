def calculate_rsi(prices, period=14):
    delta = prices.diff()
    gain = delta.where(delta > 0, 0)
    loss = -delta.where(delta < 0, 0)
    avg_gain = gain.ewm(alpha=1/period, min_periods=period, adjust=False).mean()
    avg_loss = loss.ewm(alpha=1/period, min_periods=period, adjust=False).mean()
    rs = avg_gain / avg_loss
    return 100 - (100 / (1 + rs))

def calculate_moving_averages(prices, short_window=50, long_window=200):
    ma_short = prices.rolling(short_window).mean()
    ma_long = prices.rolling(long_window).mean()
    return ma_short, ma_long

def calculate_macd(prices, fast=12, slow=26, signal=9):
    ema_fast = prices.ewm(span=fast, adjust=False).mean()
    ema_slow = prices.ewm(span=slow, adjust=False).mean()
    macd_line = ema_fast - ema_slow
    signal_line = macd_line.ewm(span=signal, adjust=False).mean()
    return macd_line, signal_line, macd_line - signal_line

flagged_corporate_action = []

for ticker in data_200.columns:

    # Corporate action guard — same check that caught VEDL

    if (daily_pct_change.abs() > 0.15).any():
        flagged_corporate_action.append(ticker)
        continue  # skip — MACD would be corrupted by the same averaging-across-the-cliff issue

    macd_line, signal_line, macd_hist = calculate_macd(prices)
    latest_macd = macd_line.iloc[-1]

def calculate_52w_high_proximity(prices, window=252):
    return (prices / prices.rolling(window).max()) * 100

def calculate_resistance_breakout(close, window=60):
    resistance = close.shift(1).rolling(window).max()
    breakout = close > resistance
    return resistance, breakout

def calculate_bollinger_bands(prices, window=20, num_std=2):
    middle_band = prices.rolling(window).mean()
    rolling_std = prices.rolling(window).std()
    upper_band = middle_band + (num_std * rolling_std)
    lower_band = middle_band - (num_std * rolling_std)
    
    # %B: where is price within the bands? 0 = at lower band, 1 = at upper band, >1 = above upper, <0 = below lower
    percent_b = (prices - lower_band) / (upper_band - lower_band)
    
    # Bandwidth: how wide are the bands relative to price? Narrow bands = low volatility, often precedes a big move
    bandwidth = (upper_band - lower_band) / middle_band * 100
    
    return middle_band, upper_band, lower_band, percent_b, bandwidth

    elif latest_percent_b > 0.8:
        signal = "Near upper band — strong/overbought zone"
    elif latest_percent_b < 0.0:
        signal = "Below lower band — strong breakdown or oversold"
    elif latest_percent_b < 0.2:
        signal = "Near lower band — weak/oversold zone"
    else:
        signal = "Middle range — no extreme"

    if bandwidth_percentile_own_history < 10:
        signal += " + SQUEEZE (unusually tight bands, watch for breakout)"

    bb_results.append({
        "ticker": ticker,
        "percent_b": round(latest_percent_b, 3),
        "bandwidth_pct": round(latest_bandwidth, 2),
        "bandwidth_percentile_own_history": round(bandwidth_percentile_own_history, 1),
        "signal": signal
    })

def plot_seasonality_heatmap(prices, ticker_name):
    monthly_ret = prices.resample("ME").last().pct_change() * 100
    monthly_ret = monthly_ret.dropna()

    df = pd.DataFrame({
        "year": monthly_ret.index.year,
        "month": monthly_ret.index.strftime("%b"),
        "return": monthly_ret.values
    })

    pivot = df.pivot(index="year", columns="month", values="return")
    month_order = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"]
    pivot = pivot.reindex(columns=[m for m in month_order if m in pivot.columns])

    fig, ax = plt.subplots(figsize=(11, max(3, len(pivot)*0.55)))
    vmax = np.nanmax(np.abs(pivot.values))
    im = ax.imshow(pivot.values, cmap="RdYlGn", vmin=-vmax, vmax=vmax, aspect="auto")

    ax.set_xticks(range(len(pivot.columns)))
    ax.set_xticklabels(pivot.columns)
    ax.set_yticks(range(len(pivot.index)))
    ax.set_yticklabels(pivot.index)

    for i in range(pivot.shape[0]):
        for j in range(pivot.shape[1]):
            val = pivot.values[i, j]
            if not np.isnan(val):
                ax.text(j, i, f"{val:.1f}", ha="center", va="center", fontsize=8,
                         color="black" if abs(val) < vmax*0.6 else "white")

    ax.set_title(f"{ticker_name} — Monthly Returns by Year (%), 10-Year History")
    plt.colorbar(im, ax=ax, label="Return (%)", shrink=0.8)
    plt.tight_layout()
    plt.show()

    # Bonus: average return by calendar month, across all years — the actual "seasonality" signal
    avg_by_month = pivot.mean(axis=0).reindex([m for m in month_order if m in pivot.columns])
    print("Average return by calendar month (across all years):")
    print(avg_by_month.round(2))

    # How many years was each month positive vs negative?
    win_rate = (pivot > 0).mean(axis=0) * 100
    print("\n% of years each month was positive:")
    print(win_rate.round(0))
